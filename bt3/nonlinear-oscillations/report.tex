\renewcommand{\maketitlehookd}{%
	\begin{abstract}
		\noindent Bài báo cáo trình bày việc áp dụng phương pháp Runge--Kutta bậc 4 (RK4) để giải hệ phương trình vi phân mô tả chuyển động dao động phi tuyến. Quá trình mô phỏng được thực hiện bằng lập trình, cho phép khảo sát sự biến thiên của các đại lượng theo thời gian. Kết quả thu được cho thấy phương pháp RK4 cho nghiệm ổn định và phản ánh rõ đặc trưng của dao động phi tuyến. Điều này khẳng định hiệu quả của phương pháp RK4 trong việc nghiên cứu các hệ động lực phức tạp.
	\end{abstract}
